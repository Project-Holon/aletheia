typedef unsigned short     u16;
typedef unsigned int       u32;
typedef signed int         s32;
typedef unsigned long long u64;

extern u32 _ll_divmod_entry_flag1(u64 value, u64 base);   /* remainder */
extern u32 _ll_divmod_entry_flag0(u64 value, u64 base);   /* quotient  */

    *--p = 0;
