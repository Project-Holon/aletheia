 * picks the band index for the distance between the object's own X

 *
 * func_0202AFFC is the identical routine for the second axis: a1[1],

 * Verified with:

typedef int            s32;
typedef unsigned int   u32;
typedef unsigned char  u8;

extern u32 func_0202B0A4(u8 *obj, s32 *pos);
extern s32 func_0202B134(u8 *obj);
extern s32 func_0202B194(s32 a, s32 b, s32 c, s32 d, s32 e);

u8 func_0202AFAC(u8 *obj, s32 *pos)
{
    u32 cur;
    u32 prev;
    s32 v;
    s32 d;
    u8 *pp;
    u8 *pc;

    cur = func_0202B0A4(obj, pos);
    if (cur == 0)
        prev = 0;
    else
        prev = (u8)(cur - 1);

    if (cur == prev)
        return (obj + cur)[0x10];

    v = pos[0] >> 12;
    d = v - func_0202B134(obj);
    if (d < 0)
        d = -d;

    pp = obj + prev;
    pc = obj + cur;

    return (u8)func_0202B194(pc[0x16] << 4, pc[0x10],
                             pp[0x16] << 4, pp[0x10], d);
}
