    byte16: and r,r8,#0xff0000   ; lsl r,r,#0x10 ; asr r,r,#0x10 ; asr r,r,#0x10

  r1 is the ifd pointer, r2 is is_last, r0 is never read (DWARF lists
  only two params).  Modelled as (void *unused, exif_ifd_data *ifd,
  short is_last); codegen-independent.
