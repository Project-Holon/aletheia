typedef unsigned char  u8;
typedef unsigned short u16;
typedef unsigned int   u32;

typedef struct {
    u16 f00;
    u16 f04;
} Ring;

extern void *memcpy(void *dst, const void *src, u32 n);

void *func_0201D768(Ring *p, u8 *src, u32 n, u32 step)
{
    u16 done = 0;
    u16 start = p->f00;
    u8 *base = (u8 *)p + 0x18;

    while (done < n) {
        u16 head = p->f00;

        memcpy(src, base + head, step);
        head = (u16)(head + step);
        if (head >= p->f04)
            head = 0;
        p->f00 = head;
        src += step;
        done = (u16)(done + step);
    }
    return base + start;
}
