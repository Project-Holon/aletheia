typedef unsigned char  u8;
typedef unsigned short u16;
typedef unsigned int   u32;

typedef struct {
    u16 f00;
    u16 f04;
} Ring;

extern void *memcpy(void *dst, const void *src, u32 n);

void *func_0201D768(Ring *p, u8 *src, u32 n, u32 step)
{
    u16 done = 0;
    u16 start = p->f00;
    u8 *base = (u8 *)p + 0x18;

    while (done < n) {
        u16 head = p->f00;

        memcpy(src, base + head, step);
        head = (u16)(head + step);
        if (head >= p->f04)
            head = 0;
        p->f00 = head;
        src += step;
        done = (u16)(done + step);
    }
    return base + start;
}

typedef unsigned short u16;
typedef unsigned int u32;
extern void MI_CpuCopy8(const void *src, void *dst, u32 size);
u8 *func_0201D768(u16 *p, const u8 *src, u16 len, u32 n)
{
  u16 total;
  u8 *base;
  u16 start;
  total = 0;
  base = (u8 *) p;
  base = base + 0x18;
  start = p[0];
  while (total < len)
  {
    u16 head = p[0];
    MI_CpuCopy8(src, base + head, n);
    src += n;
    head = (u16) (n + head);
    if (head >= p[2])

      head = 0;
    }
    p[0] = head;
    total = (u16) (total + n);
  }

  return base + start;
}
