u8 func_0202B04C(u8 *obj, s32 *pos)
{
    u32 cur;
    u32 prev;
    s32 d;
    s32 v;
    u8 *pp;
    u8 *pc;

    cur = func_0202B104(obj, pos);
    if (cur == 0)
        prev = 0;
    else
        prev = (u8)(cur - 1);

    if (cur == prev)
        return (obj + cur)[0x10];

    v = pos[2] >> 12;
    d = v - func_0202B174(obj);
    if (d < 0)
        d = -d;

    pp = obj + prev;
    pc = obj + cur;

    return (u8)func_0202B194(pc[0x22] << 4, pc[0x10],
                             pp[0x22] << 4, pp[0x10], d);
}
