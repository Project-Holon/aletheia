    {

        u32 a, b;
        s32 x, y;

        if (fmt & 0x20000000)
            return (s32)ofs;

        data += ofs;

        if ((fmt & 0xc0000000) == 0)
            goto load;

        lim = (u16)fmt;

        if (fmt & 0x40000000) {
            if (pos & 1) {
                if (pos > lim) {
                    pos = (lim >> 1) + 1;
                    goto load;
                }
                pos >>= 1;
                goto avg;
            }
            pos >>= 1;
            goto load;
        }

        if ((pos & 3) == 0) {
            pos >>= 2;
            goto load;
        }

        if (pos > lim) {
            pos = (pos & 3) + (lim >> 2);
            goto load;
        }

        if ((pos & 1) == 0) {
            pos >>= 2;
            goto avg;
        }

        if (pos & 2) {
            a = (pos >> 2) + 1;
            b = pos >> 2;
        } else {
            a = pos >> 2;
            b = (pos >> 2) + 1;
        }

        if (!(fmt & 0x10000000)) {
            x = ((s32 *)data)[a];
            y = ((s32 *)data)[b];
        } else {
            x = ((s16 *)data)[a];
            y = ((s16 *)data)[b];
        }
        x *= 3;
        x += y;
        return x >> 2;

    load:
        if (fmt & 0x10000000)
            return ((s16 *)data)[pos];
        return ((s32 *)data)[pos];

    avg:
        if (!(fmt & 0x10000000)) {
            x = ((s32 *)data)[pos];
            y = ((s32 *)data)[pos + 1];
        } else {
            x = ((s16 *)data)[pos];
            y = ((s16 *)data)[pos + 1];
        }

One shared codegen blocker, four functions, DIFF at exactly four bytes
---------------------------------------------------------------------
func_02070524, func_02070730, func_02070AFC and func_02070360 all reproduce
every branch target, every register assignment and every instruction except
one block, and each is exactly four bytes short.

In every case the original spells a two-arm conditional as

        <cond>eq  ...            /* the short arm, if-converted */
        beq       join
        <4 plain instructions>   /* the long arm, NOT if-converted */
  join:

while -O4,p if-converts BOTH arms, producing one fewer instruction:

        <4 predicated instructions, ne>
        <the short arm, eq>

The four-instruction long arm sits exactly on this front end's
if-conversion budget, and nothing tried moved it off:
