        VEC_CrossProduct(&r, &d2, &c);
        VEC_Normalize(&c, &c);
        MTX_RotAxis43(&m, &c, FX_SinCosTable_[(ang >> 4) * 2], FX_SinCosTable_[(ang >> 4) * 2 + 1]);
        r.x = rot[1];
        r.y = rot[2];
        r.z = rot[3];
        MTX_MultVec43(&r, &m, &r);
        VEC_Normalize(&r, &r);
        if (flag == 1) {
            r.x *= -1;
            r.z *= -1;
        }
        rot[1] = r.x;
        rot[2] = r.y;
        rot[3] = r.z;
        if (ang < 0x2000 || (ang > 0x6000 && ang < 0xa000) || ang > 0xe000)
            rot[0] = 0;
        func_020519E4(emitter, rot);
        func_02051880(emitter, &rot[1]);
    }

    if ((cmd->mode != 0 && (cmd->src & 1)) || cmd->mode == 0 || cmd->mode == 2 || cmd->mode == 3) {
        a = func_020517D0(emitter);
        b = func_02051830(emitter) << 12;
        e = func_02051820(emitter);
        f = func_020517E0(emitter);
        if (cmd->r44 != 0) {
            func_020517D8(emitter, FX_Mul(a, cmd->r44));
            func_020517E8(emitter, FX_Mul(f, cmd->r44));
        }
        if (cmd->r48 != 0)
            func_020518A4(emitter, FX_Mul(b, cmd->r48) >> 12);
        if (cmd->r4c != 0)
            func_0205189C(emitter, (s16)FX_Mul(e, cmd->r4c));
        if (cmd->r50 != 0) {
            func_02051790(emitter, &sc);
            sc.x = FX_Mul(sc.x, cmd->r50);
            sc.y = FX_Mul(sc.y, cmd->r50);
            sc.z = FX_Mul(sc.z, cmd->r50);
            func_020517A4(emitter, &sc);
            a = func_020517F0(emitter);
            func_02051800(emitter, FX_Mul(a, cmd->r50));
            a = func_02051808(emitter);
            func_02051818(emitter, FX_Mul(a, cmd->r50));
        }
    }
    if (cmd->mode == 3)
        func_02051838(emitter, &tmp);
    else
        func_02051838(emitter, &srcPos);
    sub_020307B0(cmd);
}
