typedef unsigned short u16;

struct DList {
    void *head;
    void *tail;
    u16   count;
    u16   node_offset;
};

struct DListNode {
    void *prev;
    void *next;
};

void func_02060480(struct DList *list, void *elem)

    node->next = 0;
    node->prev = 0;
    list->head = elem;
    list->tail = elem;
    list->count++;
}

void func_020604AC(struct DList *list, void *elem)

        func_02060480(list, elem);
        return;

        struct DListNode *elem_node = (struct DListNode *)((char *)elem + list->node_offset);
        struct DListNode *tail_node = (struct DListNode *)((char *)list->tail + list->node_offset);
        elem_node->prev = list->tail;
        elem_node->next = 0;
        tail_node->next = elem;

        list->count++;

}

void func_020604AC(volatile struct DList *list, void *elem)
{
    struct DListNode *node;
    u16               off;
    void             *tail;

    if (list->head == 0) {
        func_02060480(list, elem);
        return;
    }
    node = (struct DListNode *)((char *)elem + list->node_offset);
    node->prev = list->tail;
    node->next = 0;
