typedef unsigned char u8;
typedef signed char s8;
typedef short s16;
typedef unsigned int u32;

typedef struct AdpcmState {
    s16 pred;
    u8 index;
} AdpcmState;

extern const s8 d_021B3A0C[16];
extern const s16 d_021B3A1C[89];

static inline u8 AdpcmEncodeSample(AdpcmState *st, s16 sample)
{
    int idx;
    int pred;
    int code;
    int diff;
    int step;

    idx = st->index;
    pred = st->pred;
    step = d_021B3A1C[idx];
    diff = sample - pred;
    if (diff < 0) {
        code = 8;
        diff = -diff;
    } else {
        code = 0;
    }
    if (diff >= step) {
        code |= 4;
        diff -= step;
    }
    if (diff >= step >> 1) {
        code |= 2;
        diff -= step >> 1;
    }
    if (diff >= step >> 2) {
        code |= 1;
    }
    diff = step >> 3;
    if (code & 4) {
        diff += step;
    }
    if (code & 2) {
        diff += step >> 1;
    }
    if (code & 1) {
        diff += step >> 2;
    }
    if (code & 8) {
        diff = -diff;
    }
    pred += diff;
    if (pred < -32768) {
        pred = -32768;
    }
    if (pred > 32767) {
        pred = 32767;
    }
    idx += d_021B3A0C[code];
    if (idx < 0) {
        idx = 0;
    } else if (idx > 88) {
        idx = 88;
    }
    st->pred = (s16)pred;
    st->index = idx;
    return code;
}

void func_021AF494(u8 *dst, const s16 *src, u32 n, AdpcmState *st)
{
    u32 i;
    u32 count;
    u8 lo;
    u8 hi;

    count = n / 2;
    for (i = 0; i < count; i++) {
        lo = AdpcmEncodeSample(st, src[0]);
        hi = AdpcmEncodeSample(st, src[1]);
        src += 2;
        *dst++ = lo | (u8)(hi << 4);
    }
}
