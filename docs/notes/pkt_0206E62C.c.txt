/*
 * Residue and codegen wall:
 * Register allocation and early-exit predicating differences on 16-byte key Patricia trie lookup:
 * 1. In both functions, when count < 16, the original performs a 4-word block load:
 *      ldm r1, {r2, r3, ip, lr}
 *    allocating the four scratch registers (r2, r3, ip, lr) simultaneously and then reassigning
 *    r1 = 0 as loop counter. mwccarm splits this into ldm {r3, ip, lr} + ldr r1, [r1, #0xc]
 *    because r2 is allocated early for a hoisted return constant (0 or -1).
 * 2. In the Patricia trie descent loop:
 *      while (prev_bit > cur_bit) { ... }
 *    the original keeps the node pointer in r6 and trie base in r5 (or r4), while mwccarm
 *    permutes the callee-saved registers (swapping r6/r7/r8 roles) and predicates the early returns
 *    (moveq/popeq) rather than branching to the shared tail exit at the end of the translation unit.
 */

static inline const u32 *getTarget(const Container *c, u32 idx, u32 off) {
    if (c == 0 || idx >= c->count) {
        return 0;
    } else {
        const u8 *sub = (const u8 *)c + c->sub_offset;
        u16 keys_offset = *(const u16 *)(sub + 2);
        return (const u32 *)((const u8 *)sub + keys_offset + off);
    }
}

static inline void *getValue(const Container *c, u32 idx) {
    if (c == 0 || idx >= c->count) {
        return 0;
    } else {
        const u8 *sub = (const u8 *)c + c->sub_offset;
        u16 elem_size = *(const u16 *)sub;
        return (void *)(sub + 4 + idx * elem_size);
    }
}

WHAT THEY ARE, confirmed.  A 16-BYTE-KEY LOOKUP over a container whose record
count sits at +0x01.  Two strategies, chosen on that count:

  count < 0x10   LINEAR SCAN.  The four key words are loaded ONCE with
                 `ldm r1, {r2, r3, ip, lr}` and compared against each record
                 with the `cmp`/`ldreq`/`cmpeq` chain at 0x0206E690.
  count >= 0x10  PATRICIA TRIE DESCENT from `(u8 *)t + 8`, four bytes per
                 node: +0 the node's own bit index, +1/+2 the two children,
                 +3 the record index.  The loop runs while the PARENT's bit
                 index is greater than the CHILD's (`cmp r2, r7` / `bhi`),
                 the standard "descend while the bit index increases" test,
                 and the key bit is picked out with
                 `asr / ldr [r1, r2, lsl #2] / and #0x1f / lsr / and #1`.
                 The descent lands on one candidate, which is then compared
                 in full -- a Patricia trie only narrows, it never proves.

             (`mla r0, r2, r1, r0` -- sub + 4 + elem_size * index)
