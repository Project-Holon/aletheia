 * Register allocation and early-exit predicating differences on 16-byte key Patricia trie lookup:

 */

        u16 keys_offset = *(const u16 *)(sub + 2);
        return (const u32 *)((const u8 *)sub + keys_offset + off);
    }
}

static inline void *getValue(const Container *c, u32 idx) {

        const u8 *sub = (const u8 *)c + c->sub_offset;
        u16 elem_size = *(const u16 *)sub;
        return (void *)(sub + 4 + idx * elem_size);

}
