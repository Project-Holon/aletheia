typedef unsigned char  u8;
typedef unsigned short u16;
typedef unsigned int   u32;
typedef signed short   s16;

struct CellAnim {              /* 0x5C bytes */
    u8  pad00[0xC];
    u32 active;                /* +0x0C */
    u8  pad10[0x48];
    u32 unk58;                 /* +0x58 */
};

struct Node {                  /* 8 bytes, d->nodes[n] */
    u16 unk00;
    s16 unk02;
    s16 unk04;
    u16 unk06;
};

struct Frame {                 /* 8 bytes, e->frames[k] */
    u32 unk0;
    u16 dur;                   /* +0x04 */
    u16 unk6;
};

struct Entry {                 /* 0x10 bytes */
    u16 count;                 /* +0x00 */
    u16 unk02;
    u32 unk04;
    u32 unk08;
    struct Frame *frames;      /* +0x0C */

    u16 numNodes;              /* +0x00 */
