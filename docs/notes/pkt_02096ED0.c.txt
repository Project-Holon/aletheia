    u16 f0;

void func_02096ED0(DB *a, const DB *src, DB *b)
{

    if (b->buf[0] == 0)
        return;

    n = a->f4;
    if (n < b->f4)
        n = b->f4;
    diff = a->f2 - b->f2;
    n += diff;

        a->buf[a->f4++] = 0;

    base = a->buf;
    pa = base + n;

        pa = base + (b->f4 + diff);
    pb = b->buf + (pa - base - diff);

        pb--;

                    q--;
                } while (*q == 0);

                *q -= 1;
                q++;
                *q += 10;
            }
        }
